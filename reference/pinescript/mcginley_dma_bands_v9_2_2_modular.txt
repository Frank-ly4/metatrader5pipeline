// =====================================
// MODULE 1: CORE CALCULATIONS
// =====================================

// Volatility-Adjusted Dynamic Moving Average
f_dma(src, minLength, maxLength, atrLen) =>
    volatility = ta.atr(atrLen) / src
    clampedVolatility = math.max(0.001, volatility)
    volatilityRange = ta.highest(clampedVolatility, 100) - ta.lowest(clampedVolatility, 100)
    volatilityIndex = (clampedVolatility - ta.lowest(clampedVolatility, 100)) / math.max(volatilityRange, 1e-9)
    float dynamicLength = maxLength - (maxLength - minLength) * volatilityIndex
    int finalLength = int(math.round(dynamicLength))
    
    var float dma = na
    float alpha = 2.0 / (finalLength + 1)
    dma := na(dma[1]) ? src : alpha * src + (1 - alpha) * nz(dma[1])
    dma

// Calculate ATR Bands
f_calculateBands(upperMA, lowerMA, atr, upperOuterMult, upperInnerMult, lowerOuterMult, lowerInnerMult) =>
    upperOuter = upperMA + atr * upperOuterMult
    upperInner = upperMA + atr * upperInnerMult
    lowerInner = lowerMA - atr * lowerInnerMult
    lowerOuter = lowerMA - atr * lowerOuterMult
    [upperOuter, upperInner, lowerInner, lowerOuter]

// Trend Detection
f_detectTrend(fastDMA, slowDMA, atr, angleThreshold, useFilter) =>
    isUptrend = fastDMA > slowDMA
    slope = ta.linreg(fastDMA, 5, 0)
    normalizedSlope = slope / atr
    angleInDegrees = math.atan(normalizedSlope) * (180 / math.pi)
    isDMAFlat = math.abs(angleInDegrees) < angleThreshold
    
    isBullishTrend = useFilter ? not isDMAFlat and angleInDegrees > 0 : true
    isBearishTrend = useFilter ? not isDMAFlat and angleInDegrees < 0 : true
    
    [isUptrend, angleInDegrees, isDMAFlat, isBullishTrend, isBearishTrend]

// Regime Detection using ADX
f_detectRegime(adxLength, adxThreshold) =>
    [diPos, diNeg, adx] = ta.dmi(adxLength, adxLength)
    isTrending = adx > adxThreshold
    isRanging = not isTrending
    [isTrending, isRanging, adx]

// =====================================
// MODULE 2: ENTRY LOGIC
// =====================================

// Wait for Proof Entry Logic
f_checkEntry(highPrice, lowPrice, closePrice, openPrice, upperOuter, lowerOuter, isBullishTrend, isBearishTrend) =>
    longCrossover = ta.crossunder(lowPrice, lowerOuter)
    shortCrossover = ta.crossover(highPrice, upperOuter)
    
    longEntry = isBullishTrend and longCrossover[1] and closePrice > openPrice
    shortEntry = isBearishTrend and shortCrossover[1] and closePrice < openPrice
    
    [longEntry, shortEntry]

// Enhanced Hybrid Entry Logic
f_checkHybridEntry(highPrice, lowPrice, closePrice, openPrice, upperOuter, lowerOuter, 
                   upperInner, lowerInner, fastDMA, slowDMA,
                   angleInDegrees, angleThreshold, isBullishTrend, isBearishTrend) =>
    
    // Aggressive Entry Conditions (High Momentum)
    strongBullishMomentum = angleInDegrees > angleThreshold * 1.5  // 1.5x threshold for strong momentum
    strongBearishMomentum = angleInDegrees < -angleThreshold * 1.5
    
    // Band touches for aggressive entries
    longBandTouch = lowPrice <= lowerOuter
    shortBandTouch = highPrice >= upperOuter
    
    // Aggressive entries: Strong momentum + band touch (no wait for proof needed)
    aggressiveLongEntry = strongBullishMomentum and longBandTouch and isBullishTrend
    aggressiveShortEntry = strongBearishMomentum and shortBandTouch and isBearishTrend
    
    // Confirmation Entry Conditions (Established Trend)
    establishedUptrend = fastDMA > slowDMA
    establishedDowntrend = fastDMA < slowDMA
    
    // Pullback detection
    dmaMidline = (fastDMA + slowDMA) / 2
    bullishPullback = establishedUptrend and lowPrice <= dmaMidline and closePrice > dmaMidline
    bearishPullback = establishedDowntrend and highPrice >= dmaMidline and closePrice < dmaMidline
    
    // Confirmation entries: Established trend + pullback to midline
    confirmationLongEntry = bullishPullback and isBullishTrend
    confirmationShortEntry = bearishPullback and isBearishTrend
    
    // Original Wait for Proof logic as fallback
    longCrossover = ta.crossunder(lowPrice, lowerOuter)
    shortCrossover = ta.crossover(highPrice, upperOuter)
    waitForProofLong = isBullishTrend and longCrossover[1] and closePrice > openPrice
    waitForProofShort = isBearishTrend and shortCrossover[1] and closePrice < openPrice
    
    // Combine all entry types
    longEntry = aggressiveLongEntry or confirmationLongEntry or waitForProofLong
    shortEntry = aggressiveShortEntry or confirmationShortEntry or waitForProofShort
    
    // Return entry signals and type for debugging
    entryType = aggressiveLongEntry or aggressiveShortEntry ? "Aggressive" :
                 (confirmationLongEntry or confirmationShortEntry) ? "Confirmation" :
                 (waitForProofLong or waitForProofShort) ? "Wait for Proof" : "None"
    
    [longEntry, shortEntry, entryType]

// Ranging Market Entry Logic
f_checkRangingEntry(lowPrice, highPrice, lowerOuter, upperOuter) =>
    rangingLongEntry = ta.crossunder(lowPrice, lowerOuter)
    rangingShortEntry = ta.crossover(highPrice, upperOuter)
    [rangingLongEntry, rangingShortEntry]

// =====================================
// MODULE 3: RISK MANAGEMENT
// =====================================

// Position Sizing Calculation
f_calculatePositionSize(equity, riskPercent, price) =>
    trade_value_usd = equity * riskPercent
    qty_to_trade = trade_value_usd / price
    step_qty = syminfo.mintick
    rounded_qty = math.round(qty_to_trade / step_qty) * step_qty
    rounded_qty

// Advanced Position Sizing with Volatility Scaling
f_calculateVolatilityAdjustedSize(equity, baseRiskPercent, price, atr, atrLength) =>
    // Calculate volatility percentile
    atrPercent = atr / price
    atrPercentile = ta.percentrank(atrPercent, 100)
    
    // Scale position size inversely with volatility
    // High volatility (percentile > 80) = smaller position
    // Low volatility (percentile < 20) = larger position
    volatilityMultiplier = 1.5 - (atrPercentile / 100)
    adjustedRiskPercent = baseRiskPercent * volatilityMultiplier
    
    // Ensure we don't exceed base risk
    finalRiskPercent = math.min(adjustedRiskPercent, baseRiskPercent * 1.5)
    
    f_calculatePositionSize(equity, finalRiskPercent, price)

// Drawdown Calculation
f_calculateDrawdown(equity, initialCapital) =>
    highWaterMark = ta.highest(equity, 252) // Rolling 1-year high
    currentDrawdown = (highWaterMark - equity) / highWaterMark
    maxDrawdownReached = currentDrawdown > maxDrawdownPercent
    [currentDrawdown, maxDrawdownReached]

// Time-Based Exit Check
f_checkTimeExit(barsSinceEntry, maxBars) =>
    barsSinceEntry >= maxBars

// =====================================
// MODULE 4: EXIT LOGIC
// =====================================

// Multi-Stage Exit Management
f_manageExit(positionSize, prevPositionSize, closePrice, fastDMA, slowDMA, 
             upperOuter, upperInner, lowerInner, lowerOuter,
             prevStopLevel, prevMidlineCrossed) =>
    
    var float stopLevel = prevStopLevel
    var bool midlineCrossed = prevMidlineCrossed
    
    isNewTrade = positionSize != prevPositionSize
    
    if isNewTrade
        if positionSize > 0
            stopLevel := lowerOuter
            midlineCrossed := false
        else if positionSize < 0
            stopLevel := upperOuter
            midlineCrossed := false
    else if positionSize > 0
        dmaMidline = (fastDMA + slowDMA) / 2
        if not midlineCrossed and closePrice > dmaMidline
            midlineCrossed := true
        
        float newStop = na
        if midlineCrossed
            newStop := math.max(dmaMidline, lowerInner)
        else
            newStop := lowerOuter
        
        stopLevel := math.max(stopLevel, newStop)
    else if positionSize < 0
        dmaMidline = (fastDMA + slowDMA) / 2
        if not midlineCrossed and closePrice < dmaMidline
            midlineCrossed := true
        
        float newStop = na
        if midlineCrossed
            newStop := math.min(dmaMidline, upperInner)
        else
            newStop := upperOuter
        
        stopLevel := math.min(stopLevel, newStop)
    
    if positionSize == 0
        stopLevel := na
        midlineCrossed := false
    
    [stopLevel, midlineCrossed]

// Volatility-Adjusted Target Calculation
f_calculateVolatilityTarget(entryPrice, atr, isLong) =>
    // Target is 2-3x ATR from entry, scaled by volatility
    atrMultiplier = 2.5
    target = isLong ? entryPrice + (atr * atrMultiplier) : entryPrice - (atr * atrMultiplier)
    target

// Core Calculations
fastDMA = f_dma(dmaSource, fastMinLen, fastMaxLen, fastAtrLen)
slowDMA = f_dma(dmaSource, slowMinLen, slowMaxLen, slowAtrLen)
atr = ta.atr(atrLength)

// Trend & Regime Detection
[isUptrend, angleInDegrees, isDMAFlat, isBullishTrend, isBearishTrend] = 
     f_detectTrend(fastDMA, slowDMA, atr, trendAngleThreshold, useAngleFilter)
[isTrendingRegime, isRangingRegime, adxValue] = f_detectRegime(adxLength, adxThreshold)

// Channel MAs
float channelUpperMA = isUptrend ? fastDMA : slowDMA
float channelLowerMA = isUptrend ? slowDMA : fastDMA

// Band Calculations
[upperOuter, upperInner, lowerInner, lowerOuter] = 
     f_calculateBands(channelUpperMA, channelLowerMA, atr, 
                     upperOuterMult, upperInnerMult, lowerOuterMult, lowerInnerMult)

// --- ENTRY LOGIC EXECUTION ---

// Determine entry signals based on market regime
var trendingLongEntry = false
var trendingShortEntry = false
var trendingEntryType = "Disabled"

var rangingLongEntry = false
var rangingShortEntry = false

if useRegimeFilter
    if isTrendingRegime
        // In a trending market, use the hybrid entry logic
        [tLong, tShort, tType] = f_checkHybridEntry(high, low, close, open, upperOuter, lowerOuter,
             upperInner, lowerInner, fastDMA, slowDMA,
             angleInDegrees, trendAngleThreshold, isBullishTrend, isBearishTrend)
        trendingLongEntry := tLong
        trendingShortEntry := tShort
        trendingEntryType := tType
    else // isRangingRegime
        // In a ranging market, use the simple reversal logic
        [rLong, rShort] = f_checkRangingEntry(low, high, lowerOuter, upperOuter)
        rangingLongEntry := rLong
        rangingShortEntry := rShort
        trendingEntryType := "Ranging" // Update status
else
    // If filter is disabled, use the hybrid logic by default
    [tLong, tShort, tType] = f_checkHybridEntry(high, low, close, open, upperOuter, lowerOuter,
         upperInner, lowerInner, fastDMA, slowDMA,
         angleInDegrees, trendAngleThreshold, isBullishTrend, isBearishTrend)
    trendingLongEntry := tLong
    trendingShortEntry := tShort
    trendingEntryType := tType

// Combine final entry signals
finalLongEntry = trendingLongEntry or rangingLongEntry
finalShortEntry = trendingShortEntry or rangingShortEntry

// Position Sizing
tradeQty = useVolatilityScaling ? 
     f_calculateVolatilityAdjustedSize(strategy.equity, orderSizePercent, close, atr, atrLength) :
     f_calculatePositionSize(strategy.equity, orderSizePercent, close)

// Drawdown Check
[currentDrawdown, pauseTrading] = f_calculateDrawdown(strategy.equity, strategy.initial_capital)

// Track bars since entry
var int barsSinceEntry = 0
if strategy.position_size != 0
    if strategy.position_size != strategy.position_size[1]
        barsSinceEntry := 0
    else
        barsSinceEntry := barsSinceEntry + 1
else
    barsSinceEntry := 0

// Check time-based exit
timeExit = useTimeBasedExit and f_checkTimeExit(barsSinceEntry, maxBarsSinceEntry)

// Exit Management (using persistent variables)
var float stopLevel = na
var bool midlineCrossed = false

[newStopLevel, newMidlineCrossed] = 
     f_manageExit(strategy.position_size, strategy.position_size[1], close,
                 fastDMA, slowDMA, upperOuter, upperInner, lowerInner, lowerOuter,
                 stopLevel, midlineCrossed)

stopLevel := newStopLevel
midlineCrossed := newMidlineCrossed

// Trade Execution
if (finalLongEntry and allowLongs and strategy.position_size == 0 and not pauseTrading)
    strategy.entry("Long", strategy.long, qty=tradeQty)

if (finalShortEntry and allowShorts and strategy.position_size == 0 and not pauseTrading)
    strategy.entry("Short", strategy.short, qty=tradeQty)

// Enhanced Exit Logic
if strategy.position_size != 0
    // Calculate volatility-adjusted target
    var float entryPrice = na
    if strategy.position_size != strategy.position_size[1]
        entryPrice := close
    
    volTarget = f_calculateVolatilityTarget(entryPrice, atr, strategy.position_size > 0)
    
    // Exit with stop, target, and time-based exit
    if timeExit
        strategy.close_all("Time Exit")
    else if usePartialExits and strategy.position_size == strategy.position_size[1] * 2 // Has not taken partial yet
        strategy.exit("Partial", qty_percent=partialExitPercent, limit=volTarget, stop=stopLevel)
    else
        strategy.exit("SL", stop=stopLevel, limit=volTarget)
